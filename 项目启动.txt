# ================ Windows环境准备 ================
# 1. Python环境 (建议Python 3.8+)
# 检查Python版本
python --version

# 2. 创建虚拟环境
python -m venv venv
.\venv\Scripts\activate

# 3. 安装项目依赖
pip install -r requirements.txt
# 主要依赖包括：
# - Flask 3.0.2 及相关组件 (Flask-Talisman, Werkzeug)
# - Celery 5.3.6 和 Redis 5.0.1
# - 图像处理：Pillow 11.1.0, pytesseract 0.3.10
# - PDF处理：reportlab 4.1.0, fonttools，PyPDF2, PyMuPDF (fitz)
# - 测试工具：pytest 8.0.0, pytest-html 4.1.1
# - 其他工具：python-dotenv, python-magic, pyOpenSSL
# - Windows专用：pywin32

# 4. 安装Redis for Windows
# 下载地址：https://github.com/microsoftarchive/redis/releases
# 选择：Redis-x64-xxx.msi
# 安装时勾选：
# - Add to PATH
# - Install service

# 6. 安装Tesseract-OCR
# 下载地址：https://github.com/UB-Mannheim/tesseract/wiki
# 选择：tesseract-ocr-w64-setup-xxx.exe
# 安装时：
# - 记住安装路径（默认：C:\Program Files\Tesseract-OCR）
# - 添加到系统环境变量Path

# ================ 启动服务 ================
# 1. 启动Redis服务
# 方法1：Windows服务（推荐）
net start Redis

# 方法2：命令行启动
redis-server

# 2. 验证Redis
redis-cli ping  # 应返回PONG

# 3. 启动Celery（新终端）
# 注意：Windows必须使用 --pool=solo
.\venv\Scripts\activate
celery -A modules.tasks worker --loglevel=INFO --pool=solo

# 4. 启动Flask（新终端）
.\venv\Scripts\activate
python app.py

# ================ 功能使用指南 ================
# 1. 图片OCR处理
# - 选择"文档生成与图片处理"标签页
# - 上传图片文件
# - 勾选"对图片进行文字识别"
# - 点击"处理图片"
# - 查看识别结果，可以合并到文本区

# 2. PDF文本提取
# - 选择"提取PDF文本"标签页
# - 上传PDF文件
# - 点击"提取文本"
# - 查看提取结果，可以复制全部文本

# 3. 文档生成
# - 在文本区域输入内容
# - 点击"生成文档"
# - 等待处理完成后下载生成的PDF

# 4. 历史记录功能
# - 系统自动保存最近6条处理结果
# - 可通过"刷新"按钮更新历史记录
# - 点击单个记录右上角的删除按钮可删除该记录
# - 点击"清空"按钮可清空所有历史记录
# - 点击"复制"按钮可复制内容到剪贴板
# - 点击"插入"按钮可将内容插入到文本编辑区

# ================ 常见问题 ================
# 1. Redis服务无法启动
# - 检查服务是否已安装：services.msc
# - 检查端口占用：netstat -ano | findstr 6379
# - 重新安装Redis

# 2. Celery报错
# - 确保使用 --pool=solo
# - 检查Redis连接
# - 查看是否激活虚拟环境

# 3. OCR无法使用
# - 检查环境变量Path是否包含Tesseract路径
# - 重启终端或电脑
# - 测试命令：tesseract --version

# 4. 端口被占用
# 查找占用端口的进程：
netstat -ano | findstr 5000
# 结束进程：
taskkill /F /PID <进程ID>

# 5. 测试失败
# - 检查日志文件中的错误信息
# - 确保所有依赖都已正确安装
# - 验证配置文件是否正确
# - 检查系统权限

# 6. 历史记录不显示
# - 确认已执行过OCR或PDF提取操作
# - 检查浏览器控制台是否有错误
# - 手动点击"刷新"按钮更新历史记录

# ================ 关闭服务 ================
# 1. 关闭Flask和Celery
# 在对应终端按 Ctrl+C
# 或使用任务管理器结束python.exe进程

# 2. 关闭Redis
net stop Redis
